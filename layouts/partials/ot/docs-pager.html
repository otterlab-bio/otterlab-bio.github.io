{{/* Previous/next navigation for a manual chapter. Chapter order comes from the manual section's weight ordering, which is the same order the sidebar shows. The hub at /docs/ closes both ends of the sequence: the first chapter gets a "Documentation" back link in the previous slot, and the last chapter gets one in the next slot, so a reader never reaches a chapter with no way forward or back. */}} {{- $cur := . -}} {{- $isTw := eq site.Language.Lang "zh-tw" -}} {{- $isZh := hasPrefix site.Language.Lang "zh" -}} {{- $manual := site.GetPage "/docs/manual" -}} {{- $pages := slice -}} {{- with $manual -}} {{- range .Pages.ByWeight -}} {{- if ne .Weight 1 -}}{{- $pages = $pages | append . -}}{{- end -}} {{- end -}} {{- end -}} {{- $idx := -1 -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $cur.RelPermalink -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if ge $idx 0 -}} {{- $hub := site.GetPage "/docs" -}} {{- $prev := "" -}}{{- $next := "" -}} {{- if gt $idx 0 -}}{{- $prev = index $pages (sub $idx 1) -}}{{- end -}} {{- if lt $idx (sub (len $pages) 1) -}}{{- $next = index $pages (add $idx 1) -}}{{- end -}} {{- if or $prev $next $hub }}
{{- with $prev }} {{ cond $isTw "上一章" (cond $isZh "上一章" "Previous") }} {{ .Title }} {{- else }} {{- with $hub }} {{ cond $isTw "總覽" (cond $isZh "总览" "Overview") }} {{ .Title }} {{- end }} {{- end }} {{- with $next }} {{ cond $isTw "下一章" (cond $isZh "下一章" "Next") }} {{ .Title }} {{- else }} {{- with $hub }} {{ cond $isTw "手冊總覽" (cond $isZh "手册总览" "Documentation") }} {{ .Title }} {{- end }} {{- end }}
{{- end -}} {{- end -}}
